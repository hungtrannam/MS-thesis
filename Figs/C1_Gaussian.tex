\documentclass{standalone}
\usepackage{pgfplots}

\pgfplotsset{compat=1.11}

\usepackage[utf8]{vietnam}
\usepackage{xcolor}
\usepackage{times}
\usepackage{amsmath,amssymb,bm,cool,mathtools,amsfonts, mathptmx}						

\begin{document}
	
	\begin{tikzpicture}[scale = .75]
		\begin{groupplot}[
			group style={group size=2 by 1, horizontal sep=1.5cm}
			]
			% First subplot (1D Gaussian)
			\nextgroupplot[
			anchor=south west,
			enlargelimits=upper,
			every axis plot post/.append style={
				mark=none, domain=-4:4, samples=50, smooth
			},
			width=0.45\linewidth,
			height=0.45\linewidth,
			title={(a)},
			title style={at={(0.5,-0.15)}, anchor=north}, % đặt tiêu đề phía dưới
			]
			\addplot[color=black!45, line width=1.5pt] {1/(sqrt(2*pi))*exp(-((x)^2)/(2))};
			\node[circle,inner sep=1pt,fill=black!90,pin=350:$\mu$] at (axis cs:0,.4) {};  
			
			% Second subplot (2D Gaussian)
			\nextgroupplot[
			colormap={mymap}{color(0cm)=(black!10); color(1cm)=(black!90)},
			domain=-4:4,
			domain y=-4:4,
			samples= 50,
			samples y=50,
			colorbar,
			width=0.45\linewidth,
			height=0.45\linewidth,
			title={(b)},
			title style={at={(0.5,-0.15)}, anchor=north}, % đặt tiêu đề phía dưới
			]
			\def\centerx{2}
			\def\centery{-1}
			\addplot3[surf,domain=-2:6,domain y=-5:3,fill opacity=0.8,shader=faceted] 
			{exp(-( (x-\centerx)^2 + (y-\centery)^2)/3 )};
			\node[circle,inner sep=1pt,fill=black!90,pin=350:$\mu$] 
			at (axis cs:\centerx,\centery,1) {};
		\end{groupplot}
	\end{tikzpicture}
	
\end{document}
