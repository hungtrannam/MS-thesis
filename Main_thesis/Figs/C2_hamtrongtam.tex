\documentclass{standalone}
\usepackage{tikz}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=1.17}
\usepackage{times}

% ===== Gaussian pdf (mu, sigma) =====
\pgfmathdeclarefunction{gauss}{2}{%
	\pgfmathparse{1/(#2*sqrt(2*pi))*exp(-((x-#1)^2)/(2*#2^2))}
}

\begin{document}
	\begin{tikzpicture}
		\begin{groupplot}[
			group style={group size=3 by 1, horizontal sep=12mm},
			width=7cm, height=6cm,
			domain=-6:6, samples=400,
			ticklabel style={font=\small},
			title style={at={(0.5,-0.15)}, anchor=north}
			]
			
			% ---- 3 Gaussians ----
			\def\mA{-2} \def\sA{1}
			\def\mB{ 0} \def\sB{.6}
			\def\mC{ 2} \def\sC{1}
			
			% weights (đổi nếu cần)
			\pgfmathsetmacro{\wA}{1/3}
			\pgfmathsetmacro{\wB}{1/3}
			\pgfmathsetmacro{\wC}{1/3}
			
			% Wasserstein-2 barycenter for 1D Gaussians:
			% mu_* = sum w_i mu_i,  sigma_* = sum w_i sigma_i
			\pgfmathsetmacro{\muW}{\wA*\mA + \wB*\mB + \wC*\mC}
			\pgfmathsetmacro{\sigW}{\wA*\sA + \wB*\sB + \wC*\sC}
			
			% ===================== (a) M_L(y) = average of phi_i(y) =====================
			\nextgroupplot[title={(a)}]
			\addplot[black!50, dashed, very thick] {gauss(\mA,\sA)};
			\addplot[black!50, dashed, very thick] {gauss(\mB,\sB)};
			\addplot[black!50, dashed, very thick] {gauss(\mC,\sC)};
			\addplot[black, ultra thick, fill=black!30, opacity=.35]
			{\wA*gauss(\mA,\sA) + \wB*gauss(\mB,\sB) + \wC*gauss(\mC,\sC)};
			
					% ===================== (c) Hellinger centroid =====================
			\nextgroupplot[title={(c)}]
			\addplot[black!50, dashed, very thick] {gauss(\mA,\sA)};
			\addplot[black!50, dashed, very thick] {gauss(\mB,\sB)};
			\addplot[black!50, dashed, very thick] {gauss(\mC,\sC)};
			
			% Hellinger centroid (normalized with Z ≈ 0.5635)
			\pgfmathsetmacro{\ZH}{0.5635}
			\addplot[black, ultra thick, fill=black!30, opacity=.35]
			{ ( ( (1/3)*(sqrt(gauss(\mA,\sA)) + sqrt(gauss(\mB,\sB)) + sqrt(gauss(\mC,\sC))) )^2 ) / \ZH };
			%			
			
			% ===================== (b) M_W(y): Wasserstein-2 barycenter =====================
			\nextgroupplot[title={(b)}]
			\addplot[black!50, dashed, very thick] {gauss(\mA,\sA)};
			\addplot[black!50, dashed, very thick] {gauss(\mB,\sB)};
			\addplot[black!50, dashed, very thick] {gauss(\mC,\sC)};
			\addplot[black, ultra thick, fill=black!30, opacity=.35]
			{gauss(\muW,\sigW)};
			
	
		\end{groupplot}
	\end{tikzpicture}
\end{document}
