\section{Phần mở đầu}


\begin{frame}{1. Lý do chọn đề tài}
	\textbf{1.1. Lý do chọn đề tài}
	
	
	
	
	
	\textbf{1.2. Mục đích nghiên cứu}
	\begin{description}
		\item[Xây dựng] thuật toán phân loại và phân tích chùm cho dữ liệu PDF mất cân bằng;
		\item[Áp dụng] hai thuật toán trên cho dữ liệu hình ảnh khi các đặc trưng của nó được trích xuất và đại diện bởi các PDF để nhận dạng. 
	\end{description}
	
\end{frame}

\begin{frame}{2. Đối tượng và phạm vi}
	
	\textbf{2.1. Đối tượng nghiên cứu}
	\begin{itemize}
		\item[--] Thuật toán phân loại và phân tích chùm với dữ liệu mất cân bằng;
		\item[--] Dữ liệu hình ảnh được trích xuất thành các đối tượng đại diện.
	\end{itemize}
	
	\textbf{2.2. Phạm vi nghiên cứu}
	
	\begin{itemize}
		\item[--] Các thuật toán phân tích chùm và phân loại cho PDF;
		\item[--] Dữ liệu hình ảnh được trích xuất thành PDF đại diện;
		\item[--] Vấn đề tính toán của các phương pháp phân loại, phân tích chùm. 
	\end{itemize}
	
\end{frame}


\begin{frame}{3. Phương pháp và cấu trúc}
	
	\begin{itemize}
		\item[--] {\it Nghiên cứu lý thuyết:} Phân tích -- tổng kết, phân loại  -- hệ thống hóa kiến thức về lĩnh vực SR; So sánh đánh giá ưu điểm và nhược điểm của các thuật toán.
		\item[--] {\it Thực nghiệm khoa học:} Thực hiện các ví dụ số minh họa lý thuyết trong môi trường mô phỏng nhằm đánh giá các mô hình SR qua nhiều tiêu chí cụ thể,   từ đó xử lý so sánh kết quả bằng các phân tích thống kê.
	\end{itemize}
